create table autores (
idAuthor INT AUTO_INCREMENT PRIMARY KEY NOT NULL,
completeName VARCHAR(100) NOT NULL,
biography VARCHAR(100) NOT NULL
);

create table usuarios (
idUsuario INT AUTO_INCREMENT PRIMARY KEY NOT NULL,
username VARCHAR(100) NOT NULL,
email VARCHAR(100) NOT NULL,
password VARCHAR(100) NOT NULL
);

create table libros (
idBook INT AUTO_INCREMENT PRIMARY KEY NOT NULL,
idAuthor INT,
titleBook VARCHAR(100) NOT NULL,
releaseYear INT,
isbnCode VARCHAR(100) NOT NULL,
idUsuario INT,

FOREIGN KEY (idAuthor) REFERENCES autores(idAuthor),
FOREIGN KEY (idUsuario) REFERENCES usuarios(idUsuario)
);
